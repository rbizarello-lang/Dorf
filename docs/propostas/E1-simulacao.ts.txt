// Simulação rápida das ideias de encaixe (E1). Fora do check, sem commit: só usa src/core.
// Cada ideia é uma regra EXTRA por cima do Game/Board: pontos a mais (e, na ponte, uma
// validade diferente). O jogador guloso olha só a jogada de agora; "atento" soma o extra
// da regra à nota, "distraído" a ignora (mede quanto a regra muda as decisões).

type Ctx = { board: Board; q: number; r: number; edges: T[]; def: TileDef; rot: number; idx: number };
interface Idea {
  id: string;
  name: string;
  /** Pontos a mais antes de colocar. */
  extra?: (c: Ctx) => number;
  /** Pontos a mais depois de colocar (precisa do estado novo). */
  after?: (c: Ctx, closed: number) => number;
  /** Troca a validade (a ponte). */
  conflict?: (a: T, b: T) => boolean;
}

const nbs = (c: Ctx) => DIRS.map(([dq, dr], i) => ({ i, n: c.board.get(c.q + dq, c.r + dr) }));
const hash = (s: number, i: number) => (Math.imul(s ^ (i * 0x9e3779b1), 0x85ebca6b) >>> 7) >>> 0;

// meia-borda: 25% das bordas de terreno comum trazem uma segunda metade, vinda só da semente
function halves(def: TileDef, rot: number): (T | null)[] {
  const base: (T | null)[] = def.edges.map((t, i) => {
    if (!LAND.includes(t) || hash(def.seed, i) % 100 >= 25) return null;
    const others = LAND.filter((x) => x !== t);
    return others[hash(def.seed, i + 17) % others.length];
  });
  const out = new Array<T | null>(6).fill(null);
  for (let i = 0; i < 6; i++) out[(i + rot) % 6] = base[i];
  return out;
}
const halfMatch = (c: Ctx) => {
  const mine = halves(c.def, c.rot);
  let extra = 0, ok = 0, nb = 0, plain = 0;
  for (const { i, n } of nbs(c)) {
    if (!n) continue;
    nb++;
    const a = c.edges[i], b = n.edges[opposite(i)];
    if (a === b) { ok++; plain++; continue; }
    if (isStrict(a) || isStrict(b)) continue;
    const theirs = halves(n.def, n.rot)[opposite(i)];
    const A = [a, mine[i]], B = [b, theirs];
    if (A.some((x) => x !== null && B.includes(x))) { ok++; extra += 7 - (synergyOf(a, b) ? 5 : 0); }
  }
  if (nb >= 2 && ok === nb && plain < nb) extra += 20;
  return extra;
};

const paid = new WeakSet<object>();

interface Out { score: number; extra: number; discards: number; tiles: number }

function play(seed: number, idea: Idea, aware: boolean): Out {
  const game = new Game(seed, { ...DEFAULT_RULES, ...JSON.parse(process.env.RULES ?? '{}') });
  const board = game.board;
  const orig = board.check.bind(board);
  if (idea.conflict) {
    board.check = (q, r, edges) => {
      const c = orig(q, r, edges);
      let conflict = false;
      for (let i = 0; i < 6; i++) {
        const n = board.get(q + DIRS[i][0], r + DIRS[i][1]);
        if (n && idea.conflict!(edges[i], n.edges[opposite(i)])) conflict = true;
      }
      c.valid = !c.occupied && c.neighbors > 0 && !conflict;
      return c;
    };
  }
  const rnd = mulberry32(seed * 7919 + 13);
  let extra = 0, tiles = 0, guard = 0;
  while (!game.over && guard++ < 2000) {
    while (game.offers.length) game.choose(0);
    if (game.discardIfStuck()) continue;
    const def = game.current!;
    let best: { q: number; r: number; rot: number; s: number } | null = null;
    for (const k of board.frontier) {
      const [q, r] = unkey(k);
      for (let rot = 0; rot < 6; rot++) {
        const edges = rotateEdges(def.edges, rot);
        const c = board.check(q, r, edges);
        if (!c.valid) continue;
        let s = c.matches * 10 + (c.neighbors >= 2 && c.matches === c.neighbors ? 20 : 0) + c.synergies.length * 5 + c.influence.points + (c.site ? 30 : 0) + rnd() * 0.5;
        if (aware && idea.extra) s += idea.extra({ board, q, r, edges, def, rot, idx: board.list.length });
        if (!best || s > best.s) best = { q, r, rot, s };
      }
    }
    if (!best) break;
    game.rot = best.rot;
    const edges = rotateEdges(def.edges, best.rot);
    const ctx: Ctx = { board, q: best.q, r: best.r, edges, def, rot: best.rot, idx: board.list.length };
    const pre = idea.extra ? idea.extra(ctx) : 0;
    const res = game.place(best.q, best.r);
    if (!res) throw new Error('jogada recusada');
    tiles++;
    const post = idea.after ? idea.after(ctx, res.closed.length) : 0;
    board.score += pre + post;
    extra += pre + post;
  }
  return { score: board.score, extra, discards: game.discarded, tiles };
}

const pct = (a: number[], p: number) => { const s = [...a].sort((x, y) => x - y); return s[Math.min(s.length - 1, Math.floor(p * s.length))]; };
const mean = (a: number[]) => a.reduce((x, y) => x + y, 0) / a.length;

    const rows: Out[] = [];
    for (let s = 1; s <= N; s++) rows.push(play(s, idea, aware));

  }
}
